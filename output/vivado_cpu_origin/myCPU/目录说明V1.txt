/*--------------------------------------------------------------------*/
myCPU/ 					#myCPU总文件
  frontend/ 				#前端，负责取指、分支预测
    fetch/  					#取指
      pc.v  						#时序，<=next_pc
      npc.v 						#组合，生成next_pc
      IFport.v 						#IF流水级，有个缓存的时序，其余为组合
      IF_ID_reg.v 					#IF至ID寄存器级别，时序控制流水线
    branch/				
	  未来放分支预测（当前为空）
  backend/ 					#后端
    decode/ 					#译码，里面的都是纯组合！
      IDport.v 						#ID译码流水级
      inst_dec.v 					#激活指令标志位，整合在ID流水级
      op_dec.v 						#指令操作置1，br_op、mem_op、alu_op，整合在ID流水级
      imm_generator.v 				#立即数生成，整合在ID流水级
      get_reg_read_addr.v 			#指令操作数，寄存器堆，要地址，整合在ID流水级
      ALU_srcGenerator.v 			#alu操作数（有指令操作数、立即数等），整合在ID流水级
    execute/ 					#执行
      EXEport.v 					#EXE流水级，主体为组合，就除法有时序
      alu.v 						#alu，整合在EXE流水级里，除法有时序
      ID_EXE_reg.v 					#ID_EXE寄存器级，时序
      EXE_MEM_reg.v 				#EXE_MEM寄存器级，时序
    memory/ 					#访存
      MEMport.v 					#MEM流水级，主体为组合，就load_done_*、load_rdata_hold有时序
      MEM_WB_reg.v 					#MEM_WB寄存器级，时序
    writeback/ 					#写回
      WBport.v 						#WB流水级，纯组合
    hazard/ 					#冒险检测，纯组合
      conflict_detector.v 			#冲突检测，目前只检测数据冒险RAW写后读
      conflict_handle.v 			#冲突处理，detect->handle，分出前递、阻塞
    control/ 					#流水线控制，解决冒险
		forward_deliver.v 			#前递至ID，纯组合
		pipeline_controller.v 		#控制valid、readyGo、allowIn，实现阻塞、流水线流动，时序
  common/ 					#通用模块
    cpu_defs.vh 					#宏定义
    tools.v 						#工具模块，指独热码生成，组合
	bram_data_stream_controller.v 	#取指、访存与cpu外部内存bram交互的中间模块，保证读写时序，后续换成AXI
	regfile.v 						#寄存器堆
  top/						#顶层
    mycpu_top.v						#顶层
/*--------------------------------------------------------------------*/
